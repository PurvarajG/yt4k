# Everything yt4k needs, installed into its own venv by install.sh. Nothing
# here touches your system Python, and none of it has to be on your PATH.

# Textual 8 is the current stable major (verified 2026-08-03). The workbench
# uses its supported App, Screen, worker, and Pilot APIs.
textual>=8,<9

# The downloader itself. Kept in yt4k's venv so `git pull && ./install.sh`
# can refresh it - YouTube changes often enough that a stale yt-dlp is the
# usual reason a download suddenly stops working.
yt-dlp>=2024.1.1

# Static ffmpeg + ffprobe binaries, so clipping and transcoding work without
# Homebrew or apt. A system ffmpeg on PATH is still preferred when present
# (see yt4k/tools.py).
static-ffmpeg>=2.5
